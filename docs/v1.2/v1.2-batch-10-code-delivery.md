## 二、发布门禁的边界

1. 所有必需证据初始都是 `pending`，缺失、失败、过期、提交不匹配、摘要不匹配或 JUnit 中存在跳过都会阻断。
2. `passed` 是负责人的声明；校验器只验证声明的结构和文件一致性，**不能证明声明、审阅人身份、执行环境或证据来源真实**。需在受保护 CI、受限存储和人工审批中核对来源，必要时使用组织已有签名/证明机制。
3. 手工证据需要记录实际命令、环境、时间、产物位置、执行人、审阅人、结果、限制及对应提交；不得上传密钥、真实用户数据或未经脱敏的日志。
4. 所有 P0 必须 `resolved`；未解决 P1 必须有负责人、缓解措施、影响范围、审阅人以及未来期限。未记录风险不等于没有风险，必须另有风险审查报告。
5. 目录必须是可信且发布期间不可修改的证据目录；路径限制和摘要检查不能阻止有权限的人同时篡改清单及证据，也不是防并发篡改机制。
6. 源码检查得到 `pyproject.toml` 包版本是 `0.1.0`，API 应用版本曾显示 `1.0.0`。V1.2 是项目发布代号，**不自动等于包版本或 API 协议版本**；请以当前文件为准，由负责人审阅版本映射，不在本文擅自修改版本。

```python
from __future__ import annotations

import argparse
import hashlib
import json
import re
import subprocess
import sys
import tomllib
import xml.etree.ElementTree as ET
from datetime import UTC, datetime, timedelta
from pathlib import Path, PurePosixPath, PureWindowsPath
from typing import Any

SCHEMA_VERSION = 1
MAX_MANIFEST_BYTES = 1024 * 1024
MAX_ARTIFACT_BYTES = 8 * 1024 * 1024
MAX_EVIDENCE_AGE = timedelta(days=30)
COMMIT_PATTERN = re.compile(r"(?:[0-9a-f]{40}|[0-9a-f]{64})\Z")
SHA256_PATTERN = re.compile(r"[0-9a-f]{64}\Z")

# 类型是强制约束：JUnit 不能替代需要真实环境和人工审阅的报告。
REQUIRED_EVIDENCE = {
    "unit_regression": "junit",
    "api_smoke_contract": "junit",
    "batch04_offline_and_real_provider": "manual",
    "batch05_identity_and_approved_write": "manual",
    "batch06_revocation_and_redis_saver": "manual",
    "batch07_delivery_contract": "manual",
    "batch08_rust_boundary_and_attack_corpus": "manual",
    "batch09_recovery_deletion_and_performance": "manual",
    "cross_batch_security_and_failure": "manual",
    "windows_backend_and_rust": "manual",
    "linux_backend_and_rust": "manual",
    "python_rust_supply_chain": "manual",
    "secret_scan": "manual",
    "sbom_and_artifact_digests": "manual",
    "api_worker_container": "manual",
    "config_cors_and_deployment": "manual",
    "media_exposure_and_rollback": "manual",
    "risk_review": "manual",
}

ROOT_FIELDS = {"schema_version", "product_release", "package_version", "commit", "evidence", "risks"}
EVIDENCE_FIELDS = {"id", "kind", "status", "artifact", "sha256", "commit", "reviewer", "reviewed_at"}
RISK_FIELDS = {"id", "severity", "state", "owner", "mitigation", "scope", "reviewer", "expires_at"}


class InputError(ValueError):
    """输入或本地运行环境不符合发布校验要求。"""


def _object_without_duplicates(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for key, value in pairs:
        if key in result:
            raise InputError(f"JSON 包含重复字段：{key}")
        result[key] = value
    return result


def _reject_constant(value: str) -> None:
    raise InputError(f"JSON 不允许特殊数值：{value}")


def read_manifest(path: Path) -> dict[str, Any]:
    try:
        with path.open("rb") as stream:
            raw = stream.read(MAX_MANIFEST_BYTES + 1)
        if len(raw) > MAX_MANIFEST_BYTES:
            raise InputError("清单超过大小上限")
        value = json.loads(raw.decode("utf-8"), object_pairs_hook=_object_without_duplicates,
                           parse_constant=_reject_constant)
    except (OSError, UnicodeError, json.JSONDecodeError, RecursionError) as exc:
        raise InputError("无法读取或解析清单") from exc
    if not isinstance(value, dict):
        raise InputError("清单顶层必须是对象")
    return value


def template() -> dict[str, Any]:
    return {
        "schema_version": SCHEMA_VERSION,
        "product_release": "V1.2",
        "package_version": "",
        "commit": "",
        "evidence": [
            {"id": key, "kind": kind, "status": "pending", "artifact": "",
             "sha256": "", "commit": "", "reviewer": "", "reviewed_at": ""}
            for key, kind in REQUIRED_EVIDENCE.items()
        ],
        "risks": [],
    }


def _nonempty(value: Any) -> bool:
    return isinstance(value, str) and bool(value.strip()) and len(value) <= 512


def _parse_time(value: Any) -> datetime | None:
    if not isinstance(value, str) or len(value) > 64:
        return None
    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))
        if parsed.tzinfo is None or parsed.utcoffset() is None:
            return None
        return parsed.astimezone(UTC)
    except (ValueError, OverflowError):
        return None


def _artifact_bytes(root: Path, relative: Any) -> bytes:
    if not isinstance(relative, str) or not relative or len(relative) > 512 or "\\" in relative:
        raise InputError("证据路径必须使用相对 POSIX 路径")
    posix = PurePosixPath(relative)
    windows = PureWindowsPath(relative)
    if posix.is_absolute() or windows.is_absolute() or windows.drive or ".." in posix.parts:
        raise InputError("证据路径不允许绝对路径或上级目录")
    try:
        base = root.resolve(strict=True)
        if not base.is_dir():
            raise InputError("证据根目录不是目录")
        target = (base / Path(*posix.parts)).resolve(strict=True)
        if not target.is_relative_to(base) or not target.is_file():
            raise InputError("证据不在根目录内或不是普通文件")
        with target.open("rb") as stream:
            raw = stream.read(MAX_ARTIFACT_BYTES + 1)
    except (OSError, RuntimeError, ValueError) as exc:
        raise InputError("证据文件不存在或不可读") from exc
    if not raw:
        raise InputError("证据文件为空")
    if len(raw) > MAX_ARTIFACT_BYTES:
        raise InputError("证据文件超过大小上限")
    return raw


def _junit_passes(raw: bytes) -> bool:
    # 此门禁不接收 DTD 或实体声明。仅按 testcase 统计，避免嵌套 suite 重复计数。
    try:
        xml = raw.decode("utf-8-sig")
        if "<!DOCTYPE" in xml.upper() or "<!ENTITY" in xml.upper():
            return False
        root = ET.fromstring(xml)
    except (ET.ParseError, UnicodeError, ValueError):
        return False

    def local_name(tag: Any) -> str:
        return tag.rsplit("}", 1)[-1] if isinstance(tag, str) else ""

    if local_name(root.tag) not in {"testsuite", "testsuites"}:
        return False
    cases = [element for element in root.iter() if local_name(element.tag) == "testcase"]
    return bool(cases) and all(
        not any(local_name(child.tag) in {"failure", "error", "skipped"} for child in case)
        for case in cases
    )


def evaluate(manifest: dict[str, Any], *, root: Path, actual_commit: str,
             package_version: str, dirty: bool, now: datetime) -> list[str]:
    """返回所有阻断原因；空列表仅代表结构、一致性和显式规则通过。"""
    errors: list[str] = []
    if set(manifest) != ROOT_FIELDS:
        errors.append("清单顶层字段缺失或包含未知字段")
    if type(manifest.get("schema_version")) is not int or manifest.get("schema_version") != SCHEMA_VERSION:
        errors.append("清单 schema_version 不受支持")
    if manifest.get("product_release") != "V1.2":
        errors.append("发布代号不匹配")
    if not _nonempty(package_version) or manifest.get("package_version") != package_version:
        errors.append("包版本与 pyproject.toml 不一致")
    commit = manifest.get("commit")
    if not isinstance(commit, str) or not COMMIT_PATTERN.fullmatch(commit) or commit != actual_commit:
        errors.append("清单提交与当前完整 Git 提交不一致")
    if dirty:
        errors.append("工作区存在未提交或未跟踪文件")

    evidence = manifest.get("evidence")
    if not isinstance(evidence, list):
        errors.append("evidence 必须是数组")
        evidence = []
    seen: set[str] = set()
    for entry in evidence:
        if not isinstance(entry, dict) or set(entry) != EVIDENCE_FIELDS:
            errors.append("证据条目字段不完整或包含未知字段")
            continue
        evidence_id = entry.get("id")
        if not isinstance(evidence_id, str) or evidence_id not in REQUIRED_EVIDENCE:
            errors.append("存在未知证据 ID")
            continue
        if evidence_id in seen:
            errors.append(f"证据重复：{evidence_id}")
            continue
        seen.add(evidence_id)
        if entry.get("kind") != REQUIRED_EVIDENCE[evidence_id]:
            errors.append(f"证据类型不匹配：{evidence_id}")
        if entry.get("status") != "passed":
            errors.append(f"证据未通过：{evidence_id}")
            continue
        if entry.get("commit") != commit:
            errors.append(f"证据提交不匹配：{evidence_id}")
        if not _nonempty(entry.get("reviewer")):
            errors.append(f"证据缺少审阅人：{evidence_id}")
        reviewed = _parse_time(entry.get("reviewed_at"))
        if reviewed is None or reviewed > now or now - reviewed > MAX_EVIDENCE_AGE:
            errors.append(f"证据时间无效、在未来或已过期：{evidence_id}")
        digest = entry.get("sha256")
        if not isinstance(digest, str) or not SHA256_PATTERN.fullmatch(digest):
            errors.append(f"证据摘要格式错误：{evidence_id}")
            continue
        try:
            raw = _artifact_bytes(root, entry.get("artifact"))
        except InputError as exc:
            errors.append(f"证据不可用：{evidence_id}（{exc}）")
            continue
        if hashlib.sha256(raw).hexdigest() != digest:
            errors.append(f"证据摘要不匹配：{evidence_id}")
        if REQUIRED_EVIDENCE[evidence_id] == "junit" and not _junit_passes(raw):
            errors.append(f"JUnit 为空、失败、跳过或格式无效：{evidence_id}")

    for missing in sorted(set(REQUIRED_EVIDENCE) - seen):
        errors.append(f"缺少必需证据：{missing}")

    risks = manifest.get("risks")
    if not isinstance(risks, list):
        errors.append("risks 必须是数组")
        risks = []
    risk_ids: set[str] = set()
    for risk in risks:
        if not isinstance(risk, dict) or set(risk) != RISK_FIELDS:
            errors.append("风险条目字段不完整或包含未知字段")
            continue
        risk_id = risk.get("id")
        if not _nonempty(risk_id):
            errors.append("风险 ID 无效")
            continue
        if risk_id in risk_ids:
            errors.append(f"风险 ID 重复：{risk_id}")
        risk_ids.add(risk_id)
        severity, state = risk.get("severity"), risk.get("state")
        if severity not in ("P0", "P1", "P2") or state not in ("open", "mitigated", "resolved"):
            errors.append(f"风险等级或状态无效：{risk_id}")
            continue
        if not _nonempty(risk.get("owner")) or not _nonempty(risk.get("reviewer")):
            errors.append(f"风险缺少负责人或审阅人：{risk_id}")
        if severity == "P0" and state != "resolved":
            errors.append(f"P0 尚未解决：{risk_id}")
        if severity == "P1" and state != "resolved":
            deadline = _parse_time(risk.get("expires_at"))
            if not _nonempty(risk.get("mitigation")) or not _nonempty(risk.get("scope")):
                errors.append(f"P1 缺少缓解措施或范围：{risk_id}")
            if deadline is None or deadline <= now:
                errors.append(f"P1 期限无效或已到期：{risk_id}")
    return errors


def repository_state(repo: Path) -> tuple[str, bool, str]:
    try:
        def git(*args: str) -> str:
            result = subprocess.run(["git", "-C", str(repo), *args], check=True,
                                    capture_output=True, text=True, timeout=15)
            return result.stdout.strip()

        commit = git("rev-parse", "--verify", "HEAD")
        dirty = bool(git("status", "--porcelain", "--untracked-files=all"))
        with (repo / "pyproject.toml").open("rb") as stream:
            version = tomllib.load(stream)["project"]["version"]
        if not isinstance(version, str):
            raise InputError("项目版本不是字符串")
        return commit, dirty, version
    except (OSError, subprocess.SubprocessError, KeyError, TypeError, tomllib.TOMLDecodeError) as exc:
        raise InputError("无法读取 Git 状态或项目版本") from exc


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="V1.2 发布证据一致性校验")
    parser.add_argument("--init", type=Path, help="仅创建新的待填写清单，不覆盖已有文件")
    parser.add_argument("--manifest", type=Path, help="待检查的 JSON 清单")
    parser.add_argument("--evidence-root", type=Path, help="可信证据根目录")
    parser.add_argument("--repo", type=Path, default=Path.cwd(), help="Git 仓库根目录")
    args = parser.parse_args(argv)
    try:
        if args.init:
            if args.manifest or args.evidence_root:
                raise InputError("--init 不能与检查参数同时使用")
            with args.init.open("x", encoding="utf-8", newline="\n") as stream:
                json.dump(template(), stream, ensure_ascii=False, indent=2)
                stream.write("\n")
            print("已创建待填写模板；模板本身不代表发布通过。")
            return 0
        if not args.manifest or not args.evidence_root:
            raise InputError("检查时必须同时提供 --manifest 和 --evidence-root")
        manifest = read_manifest(args.manifest)
        commit, dirty, version = repository_state(args.repo)
        errors = evaluate(manifest, root=args.evidence_root, actual_commit=commit,
                          package_version=version, dirty=dirty, now=datetime.now(UTC))
        if errors:
            print(json.dumps({"decision": "blocked", "errors": errors}, ensure_ascii=False, indent=2))
            return 1
        print(json.dumps({"decision": "structural_checks_passed", "commit": commit}, ensure_ascii=False))
        return 0
    except (InputError, OSError) as exc:
        print(json.dumps({"decision": "invalid_input", "error": str(exc)}, ensure_ascii=False), file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
```

### 文件一详解

- `REQUIRED_EVIDENCE` 是硬编码的最低证据类别，不能通过从 JSON 中删掉一项来绕过。真实 Provider、Redis、跨批故障、安全攻击、供应链及回滚使用人工报告，因为 JUnit 中有测试成功不证明真实基础设施参与了测试。
- `read_manifest()` 拒绝重复 JSON 字段、NaN、无效 UTF-8 和过大文件；`template()` 的状态全部为 `pending`。
- `_artifact_bytes()` 将路径限制在证据根目录，读取一次有界字节，再同时用于 SHA-256 和 JUnit 解析。对路径穿越和指向根目录外的链接 fail closed；但证据目录本身仍必须可信且只读。
- `_junit_passes()` 遍历测试用例而非累加嵌套 suite 的数字，要求至少一个用例且不能有 failure、error 或 skipped。它**不验证**测试是否覆盖指定的业务点，审阅者必须对照真实测试选择、CI 命令和产物来源。
- `evaluate()` 复核提交、包版本、脏工作区、完整证据清单、审阅时间以及风险规则。30 天是本批保守默认时效，不是安全保证；若组织另有策略，应审批后同步修改代码和测试。
- `repository_state()` 读取真实 Git HEAD、包括未跟踪文件的工作区状态，以及项目元数据。证据建议放在仓库之外，否则未跟踪产物本身会让工作区变脏。忽略文件不会被 Git status 检出，构建应使用受控的干净 checkout。
- `main()` 的退出码：`0` 仅代表结构性检查通过，`1` 是发布阻断，`2` 是输入或运行环境错误。成功也不代表自动授权上线。

```python
from __future__ import annotations

import copy
import hashlib
from datetime import UTC, datetime, timedelta

import pytest

from scripts import v12_release_gate as gate

NOW = datetime(2026, 10, 28, 12, 0, tzinfo=UTC)
COMMIT = "a" * 40
VERSION = "0.1.0"
GOOD_XML = b'<testsuites><testsuite><testcase name="ok"/></testsuite></testsuites>'


@pytest.fixture
def prepared(tmp_path):
    data = gate.template()
    data["commit"] = COMMIT
    data["package_version"] = VERSION
    for entry in data["evidence"]:
        raw = GOOD_XML if entry["kind"] == "junit" else b"test-only synthetic report"
        name = entry["id"] + (".xml" if entry["kind"] == "junit" else ".txt")
        (tmp_path / name).write_bytes(raw)
        entry.update(status="passed", artifact=name, sha256=hashlib.sha256(raw).hexdigest(),
                     commit=COMMIT, reviewer="test-reviewer", reviewed_at=NOW.isoformat())
    return data, tmp_path


def check(prepared, **overrides):
    data, root = prepared
    arguments = {"root": root, "actual_commit": COMMIT, "package_version": VERSION,
                 "dirty": False, "now": NOW}
    arguments.update(overrides)
    return gate.evaluate(data, **arguments)


def item(data, evidence_id):
    return next(entry for entry in data["evidence"] if entry["id"] == evidence_id)


def risk(**changes):
    value = {"id": "R-1", "severity": "P1", "state": "open", "owner": "owner",
             "mitigation": "restrict exposure", "scope": "internal", "reviewer": "reviewer",
             "expires_at": (NOW + timedelta(days=1)).isoformat()}
    value.update(changes)
    return value


def test_synthetic_evidence_passes_only_structural_checks(prepared):
    assert check(prepared) == []
    assert check(prepared, dirty=True)
    assert check(prepared, actual_commit="b" * 40)
    assert check(prepared, package_version="different")


@pytest.mark.parametrize("state", ["pending", "failed", "unknown"])
def test_nonpassed_status_is_blocked(prepared, state):
    data, _ = prepared
    item(data, "batch06_revocation_and_redis_saver")["status"] = state
    assert any("证据未通过" in error for error in check(prepared))


def test_missing_duplicate_unknown_and_wrong_type_are_blocked(prepared):
    data, _ = prepared
    data["evidence"].pop()
```
